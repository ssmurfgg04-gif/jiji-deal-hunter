   235→function formatKES(n: number): string {
   236→  if (n >= 1000) return `${(n / 1000).toFixed(0)}K`;
   237→  return `${n}`;
   238→}
   239→
   240→function pct(n: number | null | undefined, digits = 0): string {
   241→  if (n == null) return "—";
   242→  return `${(n * 100).toFixed(digits)}%`;
   243→}
   244→
   245→function timeAgo(date: string | null): string {
   246→  if (!date) return "never";
   247→  try {
   248→    return formatDistanceToNow(new Date(date), { addSuffix: true });
   249→  } catch {
   250→    return "—";
   251→  }
   252→}
   253→
   254→function nextRunAtLabel(s: SchedulerStatus | null, inLabel: string | null): string {
   255→  if (!s) return "—";
   256→  if (s.running) return "running...";
   257→  if (!s.enabled) return "paused";
   258→  if (!s.nextRunAt) return "starting...";
   259→  return `next in ${inLabel ?? "?"}`;
   260→}

  1175→function parseFactors(l: EnrichedListing): { label: string; value: string; danger?: boolean }[] {
  1176→  if (!l.score) return [];
  1177→  const factors: Array<{ label: string; value: string; danger?: boolean }> = [
  1178→    { label: "Price vs Median", value: `${(l.score.priceVsMedian * 100).toFixed(1)}%` },
  1179→    { label: "Seller Risk", value: `${(l.score.sellerRisk * 100).toFixed(0)}%` },
  1180→    { label: "Dealer Ratio", value: l.score.dealerRatio.toFixed(1), danger: l.score.dealerRatio > 50 },
  1181→    { label: "Image Duplicates", value: String(l.score.imageDuplicateCount), danger: l.score.imageDuplicateCount > 0 },

  1185→    { label: "Ghost Listing", value: l.score.isGhostListing ? "YES" : "no", danger: l.score.isGhostListing },

  1188→    { label: "Cross-Market", value: l.score.crossMarketBroker ? "YES" : "no", danger: l.score.crossMarketBroker },
